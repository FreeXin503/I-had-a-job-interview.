### 2. backEnd/src/modules/agent/agent.service.ts
```typescript
import { AxiosResponse } from 'axios';  // 新增

async generateFirstQuestion(params: any): Promise<any> {  // 添加返回类型
  const response: AxiosResponse = await firstValueFrom(  // 添加类型注解
    this.httpService.post(
      `${this.agentApiUrl}/api/agent/generate-first-question`,
      params,
    ),
  );
  return response.data;  // 修改：之前是 response.data.data
}
```
